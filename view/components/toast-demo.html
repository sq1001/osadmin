Toast 提示示例

本页面演示 Toast 组件的使用方法，支持多种类型、位置、显示模式、消散动画等自定义配置

消息类型
显示位置
显示模式 新
堆叠模式：多条消息同时显示；替换模式：新消息替换旧消息
数量限制 新
超出数量限制时，最旧的消息会以消散动画移除
消散动画 新
消散动画：淡出 + 缩小 + 高度收缩，形成消散效果
持续时间
加载状态
自定义配置
自定义皮肤 新
通过 skin 配置项启用左侧色条皮肤（预设名），或传对象自定义样式
批量操作
回调函数

使用示例

// 基本用法（默认无标题，仅显示内容）
toast.success('操作成功！');
toast.error('操作失败！');
toast.warning('警告信息');
toast.info('提示信息');

// 仅内容，无标题无图标
toast.show('这是一条纯文本提示');

// 带标题
toast.show('操作成功', { title: '成功', icon: 'success' });

// 显示位置
toast.show('内容', { offset: 'rt' });  // 右上角
toast.show('内容', { offset: 'rb' });  // 右下角
toast.show('内容', { offset: 'lt' });  // 左上角
toast.show('内容', { offset: 'lb' });  // 左下角
toast.show('内容', { offset: 'tc' });  // 顶部居中
toast.show('内容', { offset: 'bc' });  // 底部居中
toast.show('内容', { offset: 'lc' });  // 左侧居中
toast.show('内容', { offset: 'rc' });  // 右侧居中

// 显示模式
toast.show('消息', { mode: 'stack' });    // 堆叠模式（默认）
toast.show('消息', { mode: 'replace' });  // 替换模式

// 数量限制
toast.show('消息', { maxCount: 3 });      // 最多显示 3 条
toast.show('消息', { maxCount: 0 });      // 无限制

// 消散动画
toast.show('消息', {
  dismissEffect: true,      // 启用消散动画（默认）
  dismissDuration: 400      // 消散动画时长
});
toast.show('消息', { dismissEffect: false });  // 禁用消散动画

// 自定义配置
toast.show('自定义内容', {
  title: '自定义标题',     // 标题，为空则不显示
  icon: 'success',         // 图标类型
  skin: 'success',         // 皮肤：预设名(success/error/warning/info/loading)或对象{borderLeft,iconColor,...}
  time: 3000,              // 自动关闭时间，0 不自动关闭
  offset: 'rt',            // 显示位置
  mode: 'stack',           // 显示模式
  maxCount: 5,             // 最大显示数量
  closable: false,         // 是否显示关闭按钮
  success: function(id, elem) {
    console.log('Toast已显示:', id);
  },
  cancel: function(id, elem) {
    console.log('Toast即将关闭:', id);
    return true;
  }
});

// 自定义皮肤（左侧色条）
toast.success('操作成功', { skin: 'success' });  // 绿色色条
toast.error('操作失败', { skin: 'error' });      // 红色色条
toast.warning('警告信息', { skin: 'warning' });  // 橙色色条
toast.info('提示信息', { skin: 'info' });        // 蓝色色条

// 自定义皮肤（对象格式，自定义样式）
toast.show('自定义提示', {
  skin: {
    borderLeft: '3px solid #e91e63',  // 左侧色条
    iconColor: '#e91e63'              // 图标颜色
  }
});
// 支持的属性：borderColor, borderLeft, background, color, boxShadow,
//             iconColor, titleColor, contentColor

// 加载状态
var loadId = toast.loading('加载中...');
setTimeout(function() {
  toast.dismiss(loadId);        // 关闭指定加载
}, 2000);
toast.closeLoading();           // 关闭所有加载状态
toast.closeLoading(loadId);     // 关闭指定加载状态

// 关闭操作
toast.dismiss(id);              // 消散关闭（带动画，触发 cancel 回调）
toast.close(id);                // 关闭（带动画，跳过 cancel 回调）
toast.clear();                  // 立即关闭所有（跳过回调）
toast.dismissAll();             // 消散所有（带动画，触发回调）

配置项说明

配置项 类型 默认值 说明
title String '' Toast 标题，为空则不显示标题行
icon String '' 图标类型：success/error/warning/info/loading 或自定义 HTML
skin String / Object '' 皮肤配置：字符串为预设名(success/error/warning/info/loading)启用左侧色条；对象格式可自定义样式，支持 borderColor/borderLeft/background/color/boxShadow/iconColor/titleColor/contentColor
time Number 3000 自动关闭时间，0 表示不自动关闭
offset String 'rt' 显示位置：rt/rb/lt/lb/tc/bc/lc/rc
mode String 'stack' 显示模式：stack(堆叠) / replace(替换)
maxCount Number 5 最大显示数量，0 表示无限制
dismissEffect Boolean true 是否启用消散动画
dismissDuration Number 400 消散动画时长
closable Boolean false 是否显示关闭按钮
className String '' 自定义 CSS 类名
success Function - 显示成功回调，参数
cancel Function - 关闭前回调，返回 false 可阻止关闭
